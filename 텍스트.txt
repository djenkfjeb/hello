<script>
    const URL_A = "https://docs.google.com/forms/d/e/1FAIpQLSd7cZ8KJvBmQf0ojgBBG6HXzUR4RZPmabuob-v3NBY4FPWfRA/viewform?usp=publish-editor";
    const URL_B = "https://docs.google.com/forms/d/e/1FAIpQLSeKf8m4ddBcUDRQWnVrBEGq894mS0GaQVfMWxWaGvQ2QrYxQg/viewform?usp=publish-editor";

    function getNextSurveyType() {
        const lastType = localStorage.getItem('lastSurveyType');
        if (lastType === 'A') {
            return 'B';
        } else {
            return 'A';
        }
    }

    function navigateToSurvey(event) {
        event.preventDefault(); 
        
        const nextType = getNextSurveyType();
        let targetUrl = URL_A;

        if (nextType === 'B') {
            targetUrl = URL_B;
        }

        localStorage.setItem('lastSurveyType', nextType);

        const alertBox = document.getElementById('redirect-alert');
        const fallbackLink = document.getElementById('fallback-link');
        alertBox.style.display = 'block';
        fallbackLink.href = targetUrl;

        window.open(targetUrl, '_blank');
        window.location.href = targetUrl;
    }
</script>
